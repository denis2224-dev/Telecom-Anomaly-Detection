
// Diagnostic-only Linux reproduction hook; not part of the proposed test change.
test.beforeEach(async ({page}) => {
  const client = await page.context().newCDPSession(page);
  await client.send('Emulation.setCPUThrottlingRate', { rate: 4 });
  await page.addInitScript(() => {
    (window as any).__focusEvents = [];
    const record = (event: string, element?: Element | null) => (window as any).__focusEvents.push({event, at: performance.now(), target: element?.textContent?.slice(0,80), active: document.activeElement?.textContent?.slice(0,80)});
    document.addEventListener('focusin', event => record('focusin', event.target as Element));
    document.addEventListener('keydown', event => record('keydown:'+event.key, event.target as Element));
    new MutationObserver(records => {
      for (const mutation of records) for (const node of mutation.removedNodes) {
        if (node instanceof Element && (node.textContent?.includes('Expand metric'))) record('removed-expand', node);
      }
    }).observe(document, {childList:true, subtree:true});
  });
});
test.afterEach(async ({page}, info) => {
  await info.attach('focus-events', {body: JSON.stringify(await page.evaluate(() => (window as any).__focusEvents), null, 2), contentType:'application/json'});
});
